Разработка оконных приложений PyQt5

Шпаргалка по созданию GUI на Python: заготовка окна, запуск приложения, разметка виджетов, популярные элементы интерфейса, сигналы и слоты, а также Qt Designer. Каждый раздел содержит короткий рабочий пример.

1. Быстрый старт PyQt5

PyQt5 предоставляет Python-привязки к Qt — набору инструментов для создания графических приложений.

Установка

pip install PyQt5

После установки можно импортировать классы из PyQt5.QtWidgets.

Минимальный запуск

import sys
from PyQt5.QtWidgets import QApplication

app = QApplication(sys.argv)
app.exec_()

QApplication управляет GUI-приложением и его главным циклом событий.

Базовая схема любого приложения

import sys
from PyQt5.QtWidgets import QApplication, QWidget

app = QApplication(sys.argv)

window = QWidget()
window.setWindowTitle("Моё приложение")
window.resize(500, 300)
window.show()

sys.exit(app.exec_())

2. Заготовка пустого окна QWidget

Самый простой вариант — создать объект QWidget, настроить его и показать через show().

Минимальное окно

import sys
from PyQt5.QtWidgets import QApplication, QWidget

app = QApplication(sys.argv)

window = QWidget()
window.setWindowTitle("Пустое окно")
window.resize(600, 400)
window.show()

sys.exit(app.exec_())

Полезные методы окна

window.setWindowTitle("Заголовок")
window.resize(800, 500)
window.setMinimumSize(400, 300)
window.move(100, 100)
window.show()
window.hide()
window.close()

Окно через собственный класс

class MainWindow(QWidget):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("Моё окно")
        self.resize(500, 300)

app = QApplication(sys.argv)
window = MainWindow()
window.show()
sys.exit(app.exec_())

3. Разметка виджетов — Layout

Layout автоматически размещает дочерние виджеты и помогает интерфейсу корректно изменяться при изменении размера окна.

QVBoxLayout — вертикально

from PyQt5.QtWidgets import (
    QWidget, QVBoxLayout, QPushButton
)

window = QWidget()
layout = QVBoxLayout(window)

layout.addWidget(QPushButton("Кнопка 1"))
layout.addWidget(QPushButton("Кнопка 2"))

window.show()

QHBoxLayout — горизонтально

from PyQt5.QtWidgets import (
    QWidget, QHBoxLayout, QPushButton
)

window = QWidget()
layout = QHBoxLayout(window)

layout.addWidget(QPushButton("Да"))
layout.addWidget(QPushButton("Нет"))

window.show()

QGridLayout — таблица

from PyQt5.QtWidgets import (
    QWidget, QGridLayout, QPushButton
)

window = QWidget()
layout = QGridLayout(window)

layout.addWidget(QPushButton("1"), 0, 0)
layout.addWidget(QPushButton("2"), 0, 1)
layout.addWidget(QPushButton("3"), 1, 0)

window.show()

Вложенные Layout

main = QVBoxLayout(window)
row = QHBoxLayout()

row.addWidget(QPushButton("OK"))
row.addWidget(QPushButton("Отмена"))

main.addWidget(QLabel("Выберите действие"))
main.addLayout(row)

Полезные операции Layout

КомандаНазначение
addWidget(w)Добавить виджет
addLayout(layout)Добавить вложенный layout
addStretch()Добавить растягивающееся пустое место

4. Популярные виджеты

Основные классы из PyQt5.QtWidgets и минимальные примеры их использования.

QLabel — текст

label = QLabel("Привет, PyQt5!")
layout.addWidget(label)

label.setText("Новый текст")
text = label.text()

QPushButton — кнопка

button = QPushButton("Нажми меня")
layout.addWidget(button)

button.clicked.connect(
    lambda: label.setText("Нажато!")
)

QLineEdit — строка

edit = QLineEdit()
edit.setPlaceholderText("Введите имя")
layout.addWidget(edit)

name = edit.text()
edit.clear()

QTextEdit — многострочный текст

text = QTextEdit()
text.setPlainText("Первая строка")
layout.addWidget(text)

value = text.toPlainText()
text.clear()

QCheckBox — флажок

check = QCheckBox("Согласен")
layout.addWidget(check)

if check.isChecked():
    print("Да")

QRadioButton — переключатель

male = QRadioButton("Мужской")
female = QRadioButton("Женский")

layout.addWidget(male)
layout.addWidget(female)

if male.isChecked():
    print("Выбран мужской")

QComboBox — выпадающий список

combo = QComboBox()
combo.addItems(["Python", "C++", "Java"])
layout.addWidget(combo)

print(combo.currentText())

QSpinBox — целое число

spin = QSpinBox()
spin.setRange(0, 100)
spin.setValue(10)
layout.addWidget(spin)

print(spin.value())

QSlider — ползунок

slider = QSlider(Qt.Horizontal)
slider.setRange(0, 100)
layout.addWidget(slider)

slider.valueChanged.connect(
    lambda value: print(value)
)

QListWidget — список

items = QListWidget()
items.addItems(["Анна", "Борис", "Вика"])
layout.addWidget(items)

print(items.currentItem().text())

QMessageBox — сообщение

from PyQt5.QtWidgets import QMessageBox

QMessageBox.information(
    window,
    "Готово",
    "Операция завершена!"
)

5. Сигналы и обработчики

Qt использует механизм сигналов и слотов: виджет сообщает о событии, а функция-обработчик реагирует на него.

Кнопка → функция

def say_hello():
    print("Привет!")

button.clicked.connect(say_hello)

Передача значения

def show_value(value):
    label.setText(str(value))

slider.valueChanged.connect(show_value)

Сигнал textChanged

def changed(text):
    label.setText(text)

edit.textChanged.connect(changed)

Лямбда

button.clicked.connect(
    lambda: label.setText("OK")
)

6. Qt Designer

Qt Designer позволяет визуально собирать интерфейс из готовых виджетов и сохранять его в файл .ui.

Запуск Qt Designer

designer

Если команда доступна в PATH, Qt Designer запускается из терминала. В Windows также можно запустить файл designer.exe из установленного Qt.

Что делать в Designer

  • создать Main Window или Widget;
  • перетащить виджеты на форму;
  • назначить им objectName;
  • установить Layout;
  • сохранить форму как window.ui.

Конвертация .ui в .py

pyuic5 window.ui -o ui_window.py

Для PyQt5 обычно используется утилита pyuic5. Она генерирует Python-код интерфейса из файла Designer.

Использование сгенерированного класса

import sys
from PyQt5.QtWidgets import QApplication, QMainWindow
from ui_window import Ui_MainWindow

class MainWindow(QMainWindow):
    def __init__(self):
        super().__init__()
        self.ui = Ui_MainWindow()
        self.ui.setupUi(self)

        self.ui.pushButton.clicked.connect(
            lambda: self.ui.label.setText("Привет!")
        )

app = QApplication(sys.argv)
window = MainWindow()
window.show()
sys.exit(app.exec_())

Полный маленький пример

import sys
from PyQt5.QtWidgets import *

app = QApplication(sys.argv)

window = QWidget()
window.setWindowTitle("Анкета")

layout = QVBoxLayout(window)

name = QLineEdit()
name.setPlaceholderText("Ваше имя")

button = QPushButton("Поздороваться")
label = QLabel()

layout.addWidget(name)
layout.addWidget(button)
layout.addWidget(label)

button.clicked.connect(
    lambda: label.setText(
        "Привет, " + name.text() + "!"
    )
)

window.show()
sys.exit(app.exec_())

8. Что запомнить

Логика PyQt5 в одной схеме

QApplication
      ↓
Создание окна
      ↓
Создание виджетов
      ↓
Layout → размещение виджетов
      ↓
Signals → обработчики событий
      ↓
show()
      ↓
app.exec_() → цикл событий

Главное правило

Не размещайте виджеты вручную через move(), если интерфейс можно построить с помощью Layout. Layout автоматически адаптирует интерфейс к изменению размеров окна.

Qt Designer

.ui → визуальная форма → pyuic5 → Python-код → подключение сигналов в своём классе.
