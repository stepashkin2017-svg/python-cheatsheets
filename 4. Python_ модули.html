Импорт модулей

Модуль — это файл .py с кодом. Импорт подключает его имена в вашу программу. Модуль выполняется один раз и кешируется в sys.modules.

import модуль

Обращаться к содержимому нужно через имя модуля и точку.

import math
math.sqrt(16)   # 4.0
math.pi         # 3.14159…

import … as

Псевдоним сокращает длинное имя или избегает конфликта.

import datetime as dt
dt.date(2026, 1, 1)
import numpy as np   # сторонний пакет

from модуль import имя

Берёт конкретные имена, префикс не нужен.

from math import sqrt, pi
sqrt(25)        # 5.0
from math import floor as fl
fl(2.7)         # 2

from модуль import *

Импортирует все публичные имена. Засоряет пространство имён, в рабочем коде лучше избегать.

from math import *
cos(0)          # 1.0
# какие имена придут — см. __all__

Изучение модуля

Встроенные функции помогают узнать, что внутри.

import math
dir(math)            # список имён
help(math.hypot)     # справка
math.__name__        # 'math'
math.__file__        # путь к файлу (если есть)

Где Python ищет модули

Порядок: текущая папка скрипта, затем PYTHONPATH, затем стандартные и установленные пакеты. Список — в sys.path.

import sys
sys.path           # список папок поиска
sys.path.append("/my/libs")
sys.modules.keys() # уже загруженные

Стандартная библиотека

Модули, которые идут вместе с Python — устанавливать не нужно. Ниже самые полезные с короткими примерами.

math — математика

import math
math.sqrt(9)         # 3.0
math.ceil(2.1)       # 3
math.floor(2.9)      # 2
math.factorial(5)    # 120
math.gcd(12, 18)     # 6
math.log(8, 2)       # 3.0
math.isclose(0.1+0.2, 0.3)  # True

random — случайные числа

import random
random.randint(1, 6)       # целое от 1 до 6
random.random()            # float от 0 до 1
random.choice(["a", "b"])  # один элемент
random.sample(range(10), 3)  # 3 разных
lst = [1, 2, 3]
random.shuffle(lst)        # перемешать на месте
random.seed(42)            # воспроизводимость

datetime — дата и время

from datetime import datetime, date, timedelta
now = datetime.now()
now.strftime("%d.%m.%Y %H:%M")   # '02.10.2026 14:30'
d = date(2026, 1, 15)
d + timedelta(days=30)           # 2026-02-14
datetime.strptime("2026-03-01", "%Y-%m-%d")
(date(2026, 12, 31) - d).days    # 350

json — обмен данными

import json
data = {"name": "Аня", "age": 20}
s = json.dumps(data, ensure_ascii=False)  # в строку
json.loads(s)                              # из строки
with open("u.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)
with open("u.json", encoding="utf-8") as f:
    json.load(f)

re — регулярные выражения

import re
re.findall(r"\d+", "a1 b22 c333")  # ['1', '22', '333']
re.search(r"(\w+)@(\w+)", "me@site").groups()  # ('me', 'site')
re.sub(r"\s+", " ", "a   b    c")   # 'a b c'
re.match(r"\d{3}", "123x") is not None  # True

collections — удобные структуры

from collections import Counter, defaultdict, deque
Counter("banana")              # {'a': 3, 'n': 2, 'b': 1}
Counter("banana").most_common(1)  # [('a', 3)]
d = defaultdict(list)
d["x"].append(1)               # ключ создаётся сам
q = deque([1, 2]); q.appendleft(0)  # быстрая очередь

itertools и functools

from itertools import permutations, product, chain
list(permutations([1, 2, 3], 2))  # 6 пар
list(product("ab", [1, 2]))       # [('a',1),('a',2),…]
list(chain([1], [2, 3]))          # [1, 2, 3]
from functools import lru_cache, reduce
@lru_cache(maxsize=None)
def fib(n): return n if n < 2 else fib(n-1) + fib(n-2)
reduce(lambda a, b: a * b, [1, 2, 3, 4])  # 24

Создание собственных модулей

Любой файл .py — уже модуль. Его имя без расширения становится именем для импорта.

Простой модуль

Файл geometry.py с функциями и константой.

# geometry.py
"""Геометрические расчёты."""
PI = 3.14159

def circle_area(r):
    """Площадь круга."""
    return PI * r ** 2
# main.py (в той же папке)
import geometry
geometry.circle_area(2)   # 12.56636
from geometry import PI

if __name__ == "__main__"

Блок срабатывает только при запуске файла напрямую, но не при импорте. Здесь удобно держать тесты и демо.

# tools.py
def hello(): return "hi"

if __name__ == "__main__":
    print(hello())   # при `python tools.py`
# при `import tools` — ничего не печатается

Специальные переменные

__name__   # имя модуля; '__main__' при запуске
__doc__    # докстринг модуля
__file__   # путь к файлу
__all__ = ["circle_area"]   # что отдаёт import *

Докстринги и аннотации

Строка в начале модуля и функций описывает назначение; её читает help().

def add(a: int, b: int) -> int:
    """Складывает два числа."""
    return a + b
help(add)

Перезагрузка модуля

Повторный import не перечитывает файл. Для этого есть reload (удобно в интерактивной сессии).

import importlib, geometry
importlib.reload(geometry)

Запуск как скрипта и как модуля

python geometry.py       # запуск файла
python -m geometry       # запуск как модуля
python -m pkg.sub        # модуль внутри пакета

Правила и частые ошибки

Имя файла — по правилам имён переменных: без пробелов и дефисов, не с цифры. Не называйте свой файл как стандартный модуль.

random.py    # плохо: перекроет стандартный random
my-tools.py  # нельзя импортировать
my_tools.py  # хорошо
# циклический импорт (a ↔ b) → ImportError

Пакеты

Пакет — папка с модулями. Позволяет группировать код и импортировать через точку.

Структура пакета

project/
├── main.py
└── shop/
    ├── __init__.py
    ├── cart.py
    └── pricing.py

Импорт из пакета

# main.py
import shop.cart
from shop import pricing
from shop.cart import add_item
shop.cart.add_item("чай")

__init__.py

Выполняется при импорте пакета. Можно оставить пустым или вывести нужное наружу.

# shop/__init__.py
from .cart import add_item
__all__ = ["add_item"]
# теперь: from shop import add_item

Относительные импорты

Работают внутри пакета: . — текущая папка, .. — на уровень выше.

# shop/cart.py
from .pricing import total
from . import pricing
from ..other import helper   # из соседнего пакета

Мини-проект: свой модуль в деле

Модуль textutils.py работает и как библиотека, и как скрипт.

# textutils.py
"""Утилиты для текста."""
import re
from collections import Counter

def words(text):
    return re.findall(r"\w+", text.lower())

def top(text, n=3):
    return Counter(words(text)).most_common(n)

if __name__ == "__main__":
    print(top("раз два два три три три"))   # [('три', 3), ('два', 2), ('раз', 1)]
# main.py
from textutils import top
print(top("a b a c a b", 2))   # [('a', 3), ('b', 2)]
Совет: сначала ищите нужное в стандартной библиотеке, а pip подключайте, когда готового решения в ней нет.
